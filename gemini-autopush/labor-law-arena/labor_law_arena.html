6. 실제 유저 로그인 · 1:1 협상방 · 노조 파티 서버 즉시 구축용 SQL (Supabase 호환)

-- Supabase SQL Editor에 붙여넣으면 로그인 프로필 · 채용공고 · 근로계약서 사본 · 실시간 협상방이 즉시 생성됩니다.
create table players (
  id uuid primary key default gen_random_uuid(),
  nickname text not null,
  preferred_role text check (preferred_role in ('worker', 'employer')),
  nationality text default 'KR', -- 'KR', 'E9', 'H2'
  avatar_json jsonb not null default '{}'::jsonb,
  ep_points int not null default 100, -- 무과금 성실 근속 내공 포인트
  cp_coins int not null default 0,    -- 유료 전략 코인 (강행법규 위반 시 무효 캡 적용)
  created_at timestamptz default now()
);

create table match_rooms (
  id uuid primary key default gen_random_uuid(),
  workplace_type text not null,       -- restaurant, office, factory, platform, construction, care_hospital, security_bldg, logistics
  workers_count int not null default 4,
  employer_id uuid references players(id),
  contract_json jsonb not null default '{}'::jsonb, -- 양방향 서명 및 사본 1부 교부 상태 기록
  active_event_id text,
  neg_gauge numeric default 50,
  union_formed boolean default false,
  status text default 'waiting'       -- waiting, interview, contract, negotiating, union_bargain, finished
);

# 홈페이지 월간 자동 연결 — 1단계(자동 내리기) 구현 계획

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 매달 1일 08:00(서울)에 퇴사 직원의 구성원 글과 계약 종료 업체의 자문사현황 로고 글을 라이믹스 «휴지통»으로 자동으로 내린다. 멈춤 장치와 기록을 함께 둔다.

**Architecture:** 무엇을 내릴지는 순수 로직 `functions/homepage-auto.js` 한 곳에서 고른다. 서버(`functions/index.js`)는 자료를 읽고, 라이믹스에 로그인하고, 휴지통으로 보내는 일만 한다. 연결(글 ↔ sid / companyId)은 화면(`pu-home.html`)에서 사람이 한 번 확인해 넣는다. 자동은 연결된 것만 움직인다.

**Tech Stack:** Firebase Functions v1(`functions.runWith().pubsub.schedule` / `https.onRequest`), RTDB, node:test, 라이믹스 문서 관리(`procDocumentManageCheckedDocument`, type=trash).

**Spec:** `docs/superpowers/specs/2026-10-05-홈페이지-월간-자동연결-design.md`

## Global Constraints

- 내리는 `type` 은 `HW.내리는type`(= "trash") 하나뿐이다. `delete`·`move`·`copy` 를 보낼 길을 만들지 않는다.
- 휴직(`status:"leave"`)은 내리지 않는다. `retired` 만 퇴사다.
- 이름으로 잇지 않는다. 자동은 `homepage/members/{key}.sid`, `homepage/partners/{companyId}.boardSrl` 로 연결된 것만 움직인다.
- 멈춤 문턱은 `멈춤문턱 = 5` 이다. 넘으면 0건을 내린다.
- ERP(`data/user_dir`·`data/companies`)는 읽기만 한다. 비었거나 못 읽으면 실패로 끝낸다.
- 게시판 허용 목록은 `people_board`·`partner_board` 둘뿐이다.
- 서버 입구는 `uid_roles/{uid}.isAdmin === true` 일 때만 연다.
- 검사는 지금 값을 박지 않는다(CLAUDE.md). 새 규칙마다 변이 검증을 한다. 주석은 걷고 본다.
- 화면: 한 칸은 한 줄, 27색 팔레트 안에서만 쓴다, 예시 이름은 홍길동·가나상사.
- 커밋은 경로를 찍는다(`git add .` 금지).

```js
"use strict";
/* 홈페이지 월간 자동 연결 — «무엇을 내릴지» 고르는 순수 로직 (설계 2026-10-05)
   ★ 고르는 규칙은 여기 «한 곳»에만 둔다. 서버(월간·지금 돌리기)와 화면 미리보기가 모두 이것을 쓴다.
   ★ 이름으로 잇지 않는다 — sid / companyId 로 «사람이 확인해 이어 둔» 글만 고른다.
   ⚠ 휴직(leave)은 퇴사가 아니다. retired 만 내린다. */

const 멈춤문턱 = 5;
const 게시판 = { 구성원: "people_board", 자문사: "partner_board" };

function 목록으로(raw) {
  let l = (raw && raw.v !== undefined) ? raw.v : raw;
  if (l && !Array.isArray(l) && typeof l === "object") l = Object.keys(l).map((k) => l[k]);
  return Array.isArray(l) ? l.filter((x) => x && typeof x === "object") : null;
}
function 번호(n) { const x = Number(n); return Number.isInteger(x) && x > 0 ? x : 0; }
/* 사유가 있는 남기기만 남기기다 — js/pu-home-diff.js keepOnSiteReason 과 같은 잣대 */
function 남기기있나(k) { return !!(k && typeof k === "object" && String(k.why || "").trim()); }
function 지문(목록) {
  return (목록 || []).map((x) => x.게시판 + ":" + x.srl).sort().join(",");
}
function 빈결과(실패) {
  return { ok: false, 실패: 실패, 내릴것: [], 보낼것: [], 연결확인: [], 멈춤: "", 지문: "" };
}

function 고르기(자료) {
  const d = 자료 || {};
  const 명부 = 목록으로(d.roster);
  const 업체 = 목록으로(d.companies);
  /* ⚠ 비어 있는 것도 «못 읽은 것»으로 본다 — 2026-09-29 처럼 자료가 한꺼번에 비어 보이면
       모두 퇴사·종료로 읽혀 홈페이지를 통째로 비우게 된다. */
  if (!명부 || !명부.length) return 빈결과("푸른ERP 직원 명부를 읽지 못했습니다");
  if (!업체 || !업체.length) return 빈결과("푸른ERP 업체관리를 읽지 못했습니다");

  const 사람 = {};
  명부.forEach((p) => { if (p.sid) 사람[String(p.sid)] = p; });
  const 회사 = {};
  업체.forEach((c) => { if (c.id && !c._deleted) 회사[String(c.id)] = c; });

  const 내릴것 = [], 연결확인 = [];

  const members = (d.members && typeof d.members === "object") ? d.members : {};
  const 구성원번호셈 = {};
  Object.keys(members).forEach((k) => {
    const s = 번호((members[k] || {}).srl);
    if (s) 구성원번호셈[s] = (구성원번호셈[s] || 0) + 1;
  });
  Object.keys(members).forEach((k) => {
    const m = members[k] || {};
    const srl = 번호(m.srl);
    if (!srl || m.takenDown) return;
    if (!m.sid) { 연결확인.push({ 종류: "구성원", key: k, srl: srl, 까닭: "직원과 이어지지 않음" }); return; }
    const p = 사람[String(m.sid)];
    if (!p) { 연결확인.push({ 종류: "구성원", key: k, srl: srl, 까닭: "명부에서 찾지 못함" }); return; }
    if (String(p.status || "") !== "retired") return;
    if (남기기있나(m.keepOnSite)) return;
    if (구성원번호셈[srl] > 1) { 연결확인.push({ 종류: "구성원", key: k, srl: srl, 까닭: "글 번호가 겹침" }); return; }
    내릴것.push({ 종류: "퇴사", 게시판: 게시판.구성원, srl: srl, key: k, sid: String(m.sid) });
  });

  const partners = (d.partners && typeof d.partners === "object") ? d.partners : {};
  const 로고번호셈 = {};
  Object.keys(partners).forEach((id) => {
    const s = 번호((partners[id] || {}).boardSrl);
    if (s) 로고번호셈[s] = (로고번호셈[s] || 0) + 1;
  });
  Object.keys(partners).forEach((id) => {
    const p = partners[id] || {};
    const srl = 번호(p.boardSrl);
    if (!srl || p.takenDown) return;
    const c = 회사[String(id)];
    if (!c) { 연결확인.push({ 종류: "자문사", companyId: id, srl: srl, 까닭: "업체관리에서 찾지 못함" }); return; }
    if (String(c.status || "") !== "closed") return;
    if (남기기있나(p.keep)) return;
    if (로고번호셈[srl] > 1) { 연결확인.push({ 종류: "자문사", companyId: id, srl: srl, 까닭: "글 번호가 겹침" }); return; }
    내릴것.push({ 종류: "계약종료", 게시판: 게시판.자문사, srl: srl, companyId: String(id) });
  });

  내릴것.sort((a, b) => (a.게시판 < b.게시판 ? -1 : a.게시판 > b.게시판 ? 1 : a.srl - b.srl));
  const 멈춤 = 내릴것.length > 멈춤문턱
    ? "한 번에 " + 내릴것.length + "건 — " + 멈춤문턱 + "건이 넘어 하나도 내리지 않았습니다" : "";
  return { ok: true, 실패: "", 내릴것: 내릴것, 보낼것: 멈춤 ? [] : 내릴것.slice(),
           연결확인: 연결확인, 멈춤: 멈춤, 지문: 지문(내릴것) };
}

module.exports = { 멈춤문턱, 게시판, 고르기, 지문, 남기기있나 };
```

```js
  const 쓴것 = [], 붙인것 = [], 내린것 = [];
  return {
    쓴것, 붙인것, 내린것,
    도구: {
```

# 정부지원금(govaid) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 지원사업 사전(조건 구조화)으로 기존·예비 고객사마다 지원사업 «후보»를 자동 계산하고, 노무사가 해당·보류·비해당을 기록하는 새 포털 앱 `pu-govaid.html` 을 만든다.

**Architecture:** 판정은 순수 함수 엔진(`js/pu-govaid-match.js`)이, 화면 묶음(목록·표·사업별)은 순수 함수 뷰 모듈(`js/pu-govaid-view.js`)이 만든다. 화면(`pu-govaid.html`)은 읽기(`govaid`·`data/companies`·`pubdata/co`)와 DOM 그리기, 그리고 공용 저장 관문(강제 모드)을 통한 쓰기만 한다. 엔진 결과는 저장하지 않고 사람의 결정(`govaid/reviews`)만 저장한다.

**Tech Stack:** 단일 HTML + 바닐라 JS(브라우저·Node 겸용 UMD 모듈), Firebase RTDB compat 9.23.0, `node:test`, 공용 모듈 `js/pu-ontology.js`·`js/pu-ontology-write.js`·`js/pu-gate.js`·`js/pu-whoami.js`.

```js
/* 정부지원금 판정 엔진 — 설계 5절의 판정 규칙을 못 박는다 (2026-10-11)
   ★ 이 검사는 «규칙»을 본다: 초안 제외 · 지난해 확정 = 초안 · 모름 ≠ 0 · 0명은 값 · 안 맞음이 모름을 이긴다. */
const test = require('node:test');
const assert = require('node:assert/strict');
const M = require('../js/pu-govaid-match.js');

const Y = 2026;
function prog(cond, over) {
  return Object.assign({ id: 'gp-t', name: '시험 사업', status: 'confirmed', baseYear: Y, cond: cond }, over || {});
}
function pf(vals, facts) {
  const v = {};
  ['headcount', 'foundedYear', 'size', 'region', 'industry', 'salesDropPct'].forEach((k) => {
    v[k] = { v: vals && vals[k] !== undefined ? vals[k] : null, src: '시험' };
  });
  const f = {};
  Object.keys(facts || {}).forEach((k) => { f[k] = { v: facts[k], src: '시험' }; });
  return { key: 'co_x', kind: 'co', id: 'x', name: '(예시)회사', bizno: '', aliases: [], vals: v, facts: f };
}

test('초안 사업은 계산하지 않는다 (조건 미확정)', () => {
  const r = M.judge(pf({ headcount: 10 }), prog({ headcount: { min: 5 } }, { status: 'draft' }), Y);
  assert.equal(r.verdict, 'unconfirmed');
});

test('지난해 확정은 올해 계산에서 초안 취급 — 자료는 그대로', () => {
  const p = prog({ headcount: { min: 5 } }, { baseYear: Y - 1 });
  assert.equal(M.judge(pf({ headcount: 10 }), p, Y).verdict, 'unconfirmed');
  assert.equal(p.status, 'confirmed', '판정이 자료를 고쳐 쓰면 안 된다');
});

test('조건이 하나도 없는 확정 사업도 계산하지 않는다 — 모든 회사가 후보로 쏟아지지 않게', () => {
  assert.equal(M.judge(pf({ headcount: 10 }), prog({}), Y).verdict, 'unconfirmed');
});

test('값이 없으면 «모름» → 정보 부족', () => {
  const r = M.judge(pf({}), prog({ headcount: { min: 5 } }), Y);
  assert.equal(r.verdict, 'unknown');
  assert.equal(r.items[0].state, 'unknown');
});

test('0 은 값이다 — 0명은 «모름»이 아니다', () => {
  assert.equal(M.judge(pf({ headcount: 0 }), prog({ headcount: { min: 1 } }), Y).verdict, 'no');
  assert.equal(M.judge(pf({ headcount: 0 }), prog({ headcount: { max: 10 } }), Y).verdict, 'candidate');
});

test('안 맞는 조건이 하나라도 있으면 «안 맞음» — 모름보다 먼저', () => {
  const r = M.judge(pf({ headcount: 3 }), prog({ headcount: { min: 5 }, regions: ['충남'] }), Y);
  assert.equal(r.verdict, 'no');
});

test('사람이 볼 조건은 후보여도 «사람 확인»으로 붙는다', () => {
  const r = M.judge(pf({ headcount: 9 }), prog({ headcount: { min: 5 }, manual: ['경영상 이유'] }), Y);
  assert.equal(r.verdict, 'candidate');
  assert.ok(r.items.some((i) => i.state === 'manual'));
});

test('업력은 개업 연도로 센다', () => {
  assert.equal(M.judge(pf({ foundedYear: Y - 3 }), prog({ yearsInBiz: { max: 7 } }), Y).verdict, 'candidate');
  assert.equal(M.judge(pf({ foundedYear: Y - 9 }), prog({ yearsInBiz: { max: 7 } }), Y).verdict, 'no');
});

test('업종 — 해당 목록 밖이거나 제외 목록 안이면 안 맞음', () => {
  assert.equal(M.judge(pf({ industry: 'C' }), prog({ industries: { include: ['C'] } }), Y).verdict, 'candidate');
  assert.equal(M.judge(pf({ industry: 'F' }), prog({ industries: { include: ['C'] } }), Y).verdict, 'no');
  assert.equal(M.judge(pf({ industry: 'K' }), prog({ industries: { exclude: ['K'] } }), Y).verdict, 'no');
});

test('예/아니오 사실 — 답이 없으면 모름, 원하는 답과 다르면 안 맞음', () => {
  const p = prog({ facts: [{ key: 'retireAgePolicy', want: true }] });
  assert.equal(M.judge(pf({}, {}), p, Y).verdict, 'unknown');
  assert.equal(M.judge(pf({}, { retireAgePolicy: true }), p, Y).verdict, 'candidate');
  assert.equal(M.judge(pf({}, { retireAgePolicy: false }), p, Y).verdict, 'no');
  const notWanted = prog({ facts: [{ key: 'hasFixedTerm', want: false }] });
  assert.equal(M.judge(pf({}, { hasFixedTerm: false }), notWanted, Y).verdict, 'candidate');
});

test('파이어베이스가 배열을 객체로 돌려줘도 같게 판정한다', () => {
  const asArr = prog({ regions: ['충남', '세종'] });
  const asObj = prog({ regions: { 0: '충남', 1: '세종' } });
  assert.equal(M.judge(pf({ region: '세종' }), asArr, Y).verdict, M.judge(pf({ region: '세종' }), asObj, Y).verdict);
  assert.deepEqual(M.list({ 0: 'a', 1: 'b' }), ['a', 'b']);
  assert.deepEqual(M.list(null), []);
});

test('주소 → 시·도 짧은 이름 (모르면 빈 글자)', () => {
  assert.equal(M.regionOf('충청남도 천안시 동남구 …'), '충남');
  assert.equal(M.regionOf('(31116) 충남 천안시'), '충남');
  assert.equal(M.regionOf('세종특별자치시 한누리대로'), '세종');
  assert.equal(M.regionOf('광주광역시 북구'), '광주');
  assert.equal(M.regionOf('광주시 오포읍'), '', '경기 광주시는 광주광역시와 헷갈리므로 모름');
  assert.equal(M.regionOf(''), '');
});

test('국민연금·고용보험 인원 — 자료가 있을 때만 값, 빈 자료의 0 은 모름', () => {
  const nps = M.headcountAuto({ nps: { now: { ym: '202609', cnt: 42 } } });
  assert.equal(nps.v, 42);
  assert.match(nps.src, /국민연금/);
  assert.equal(M.headcountAuto({ nps: { now: { ym: '202609', cnt: 0 } } }).v, 0, '자료가 있는 0 은 값');
  assert.equal(M.headcountAuto({ ei: { rows: null, gy: 0 } }).v, null, '공단에 없는 회사의 gy:0 은 모름');
  assert.equal(M.headcountAuto({ ei: { rows: [{ fg: 'gy', cnt: 7 }], gy: 7 } }).v, 7);
  assert.equal(M.headcountAuto(null).v, null);
});

test('업체 값 — 직접 넣은 값이 이기고, 업체관리 «기타» 규모는 모름', () => {
  const p = M.profileFromCompany({ id: 'c1', name: '(예시)회사', bizNo: '123-45-67890', address: '충남 아산시', companySize: '기타' },
    { nps: { now: { ym: '202609', cnt: 30 } } }, { vals: { headcount: 35 } });
  assert.equal(p.key, 'co_c1');
  assert.equal(p.bizno, '1234567890');
  assert.deepEqual(p.vals.headcount, { v: 35, src: '직접 입력' });
  assert.equal(p.vals.region.v, '충남');
  assert.equal(p.vals.size.v, null);
});

test('예비 고객 값 — 입력한 값을 쓰고 비면 모름', () => {
  const p = M.profileFromProspect({ id: 'p1', name: '(예시)예비', bizno: '', region: '세종', headcount: '' }, null, null);
  assert.equal(p.key, 'pr_p1');
  assert.equal(p.vals.region.v, '세종');
  assert.equal(p.vals.headcount.v, null);
});

test('서울 시각으로 해를 센다 — 12월 31일 15시(UTC)는 서울로 이미 새해', () => {
  assert.equal(M.kstYear(Date.UTC(2026, 11, 31, 15, 0, 0)), 2027);
  assert.equal(M.kstYear(Date.UTC(2026, 11, 31, 14, 0, 0)), 2026);
});
```

```js
    K: '금융·보험업', L: '부동산업', M: '전문·과학·기술 서비스업', N: '사업시설관리·사업지원·임대 서비스업',
    O: '공공행정·국방·사회보장', P: '교육 서비스업', Q: '보건업·사회복지 서비스업', R: '예술·스포츠·여가',
    S: '협회·단체·수리·기타 개인 서비스업', T: '가구 내 고용활동', U: '국제·외국기관'
  };
  /* 사람만 답할 수 있는 «예/아니오» 사실 — 사업마다 이름이 갈라지지 않게 «이 목록에서만» 고른다 */
  var FACTS = {
    prioritySupport: '우선지원대상기업이다',
    smeCert: '중소기업 확인서가 있다',
    retireAgePolicy: '정년 제도가 있다',
    hasWorkers60: '만 60세 이상 근로자가 있다',
    hasFixedTerm: '기간제 근로자가 있다',
    hiredYouth: '최근 청년을 채용했다',
    flexWork: '유연근무·근로시간 단축을 운영한다'
  };
  /* 주소 앞머리 → 시·도 짧은 이름. ⚠ 「광주시」(경기 광주)는 광주광역시와 헷갈리므로 regionOf 가 먼저 걸러 모름으로 둔다 */
  var REGION_PREFIX = [
    ['서울', '서울'], ['부산', '부산'], ['대구', '대구'], ['인천', '인천'], ['광주', '광주'],
    ['대전', '대전'], ['울산', '울산'], ['세종', '세종'], ['경기', '경기'], ['강원', '강원'],
    ['충북', '충청북도'], ['충북', '충북'], ['충남', '충청남도'], ['충남', '충남'],
    ['전북', '전라북도'], ['전북', '전북'], ['전남', '전라남도'], ['전남', '전남'],
    ['경북', '경상북도'], ['경북', '경북'], ['경남', '경상남도'], ['경남', '경남'], ['제주', '제주']
  ];
  var MINE = '직접 입력';

  function clean(v) { return v == null ? '' : String(v).trim(); }
  function digits(v) { return clean(v).replace(/[^0-9]/g, ''); }
  function blank(v) {
    return v == null || (typeof v === 'string' && v.trim() === '') || (typeof v === 'number' && !isFinite(v));
  }
  function num(v) { if (blank(v)) return null; var n = Number(v); return isFinite(n) ? n : null; }
  /* 파이어베이스는 배열을 {0:…,1:…} 로 돌려줄 때가 있다 — 둘 다 배열로 */
  function list(v) {
    var out = Array.isArray(v) ? v : (v && typeof v === 'object' ? Object.keys(v).map(function (k) { return v[k]; }) : []);
    return out.filter(function (x) { return x != null; });
  }
  function kstYear(now) { return new Date((now == null ? Date.now() : now) + 9 * 3600e3).getUTCFullYear(); }

  function regionOf(address) {
    var a = clean(address).replace(/^\(?\d{5}\)?\s*/, '');
    if (/^광주시/.test(a)) return '';
    for (var i = 0; i < REGION_PREFIX.length; i++) if (a.indexOf(REGION_PREFIX[i][1]) === 0) return REGION_PREFIX[i][0];
    return '';
  }

  /* 후보 [값, 출처] 가운데 «빈 것이 아닌» 첫 값 — 0 은 빈 것이 아니다 */
  function first(cands) {
    for (var i = 0; i < cands.length; i++) if (!blank(cands[i][0])) return { v: cands[i][0], src: cands[i][1] };
    return { v: null, src: '' };
  }
  /* pubdata/co/{사업자번호} — nps-monthly.js 가 담는다. ⚠ 공단에 없는 회사는 ei.rows 가 비고 gy 가 0 이다 → 모름 */
  function headcountAuto(pub) {
    pub = pub || {};
    var now = pub.nps && pub.nps.now, ym = clean(now && now.ym);
    if (ym && num(now.cnt) != null) return { v: num(now.cnt), src: '국민연금 가입자 ' + ym.slice(0, 4) + '-' + ym.slice(4, 6) };
    var ei = pub.ei;
    if (ei && list(ei.rows).length && num(ei.gy) != null) return { v: num(ei.gy), src: '고용보험 상시인원' };
    return { v: null, src: '' };
  }
  function factsOf(extra) {
    var f = (extra && extra.facts) || {}, out = {};
    Object.keys(FACTS).forEach(function (k) { if (typeof f[k] === 'boolean') out[k] = { v: f[k], src: MINE }; });
    return out;
  }
  function sizeOk(v) { return SIZES.indexOf(clean(v)) >= 0 ? clean(v) : ''; }
  function regionOk(v) { return REGIONS.indexOf(clean(v)) >= 0 ? clean(v) : ''; }
  function industryOk(v) { return Object.prototype.hasOwnProperty.call(INDUSTRIES, clean(v)) ? clean(v) : ''; }

  /* 기존 고객사 — 업체관리 원장(data/companies/v 한 줄) + 공공데이터 + 보충값(govaid/extras/co_{id}) */
  function profileFromCompany(co, pub, extra) {
    co = co || {};
    var ex = (extra && extra.vals) || {}, hc = headcountAuto(pub);
    return {
      key: 'co_' + clean(co.id), kind: 'co', id: clean(co.id), name: clean(co.name), bizno: digits(co.bizNo), aliases: [],
      vals: {
        headcount: first([[num(ex.headcount), MINE], [hc.v, hc.src]]),
        foundedYear: first([[num(ex.foundedYear), MINE]]),
        size: first([[sizeOk(ex.size), MINE], [sizeOk(co.companySize), '업체관리']]),
        region: first([[regionOk(ex.region), MINE], [regionOf(co.address), '업체관리 주소']]),
        industry: first([[industryOk(ex.industry), MINE]]),
        salesDropPct: first([[num(ex.salesDropPct), MINE]])
      },
      facts: factsOf(extra)
    };
  }
  /* 예비 고객 — govaid/prospects/{id} + 공공데이터(사업자번호가 있으면) + 보충값(govaid/extras/pr_{id}) */
  function profileFromProspect(pr, pub, extra) {
    pr = pr || {};
    var ex = (extra && extra.vals) || {}, hc = headcountAuto(pub), IN = '예비 고객 입력';
    return {
      key: 'pr_' + clean(pr.id), kind: 'pr', id: clean(pr.id), name: clean(pr.name), bizno: digits(pr.bizno), aliases: [],
      vals: {
        headcount: first([[num(ex.headcount), MINE], [num(pr.headcount), IN], [hc.v, hc.src]]),
        foundedYear: first([[num(ex.foundedYear), MINE], [num(pr.foundedYear), IN]]),
        size: first([[sizeOk(ex.size), MINE], [sizeOk(pr.size), IN]]),
        region: first([[regionOk(ex.region), MINE], [regionOk(pr.region), IN]]),
        industry: first([[industryOk(ex.industry), MINE], [industryOk(pr.industry), IN]]),
        salesDropPct: first([[num(ex.salesDropPct), MINE]])
      },
      facts: factsOf(extra)
    };
  }

  function hasRange(r) { return !!r && (num(r.min) != null || num(r.max) != null); }
  function inRange(r, v) { var lo = num(r.min), hi = num(r.max); return (lo == null || v >= lo) && (hi == null || v <= hi); }
  function rangeText(r, unit) {
    var lo = num(r.min), hi = num(r.max);
    if (lo != null && hi != null) return lo + '~' + hi + unit;
    return lo != null ? lo + unit + ' 이상' : hi + unit + ' 이하';
  }
  function indName(k) { return INDUSTRIES[k] || k; }

  function judge(profile, program, year) {
    program = program || {}; profile = profile || {};
    var vals = profile.vals || {}, facts = profile.facts || {}, c = program.cond || {}, items = [];
    if (program.status !== 'confirmed' || !(num(program.baseYear) >= num(year))) return { verdict: 'unconfirmed', items: items };
    function v(k) { return (vals[k] || {}).v; }
    function add(key, label, state) { items.push({ key: key, label: label, state: state }); }
    function range(key, r, val, label, unit) {
      if (!hasRange(r)) return;
      var text = label + ' ' + rangeText(r, unit);
      if (val == null) add(key, text, 'unknown'); else add(key, text, inRange(r, val) ? 'ok' : 'no');
    }
    function oneOf(key, arr, val, label) {
      arr = list(arr); if (!arr.length) return;
      var text = label + ': ' + arr.join('·');
      if (blank(val)) add(key, text, 'unknown'); else add(key, text, arr.indexOf(val) >= 0 ? 'ok' : 'no');
    }
    range('headcount', c.headcount, num(v('headcount')), '근로자', '명');
    var fy = num(v('foundedYear'));
    range('yearsInBiz', c.yearsInBiz, fy == null ? null : num(year) - fy, '업력', '년');
    range('salesDropPct', c.salesDropPct, num(v('salesDropPct')), '매출 감소', '%');
    oneOf('size', c.sizes, v('size'), '규모');
    oneOf('region', c.regions, v('region'), '지역');
    var ind = c.industries || {}, inc = list(ind.include), exc = list(ind.exclude), iv = v('industry');
    if (inc.length || exc.length) {
      var text = '업종' + (inc.length ? ': ' + inc.map(indName).join('·') : '') + (exc.length ? ' (제외 ' + exc.map(indName).join('·') + ')' : '');
      if (blank(iv)) add('industry', text, 'unknown');
      else add('industry', text, (inc.length && inc.indexOf(iv) < 0) || exc.indexOf(iv) >= 0 ? 'no' : 'ok');
    }
    list(c.facts).forEach(function (f) {
      if (!f || !FACTS[f.key]) return;
      var want = f.want !== false, got = facts[f.key];
      var label = FACTS[f.key] + (want ? '' : ' — 아니어야 함');
      if (!got || typeof got.v !== 'boolean') add('fact:' + f.key, label, 'unknown');
      else add('fact:' + f.key, label, got.v === want ? 'ok' : 'no');
    });
    list(c.manual).forEach(function (m, i) { if (clean(m)) add('manual:' + i, clean(m), 'manual'); });
    if (!items.length) return { verdict: 'unconfirmed', items: items };
    var states = items.map(function (x) { return x.state; });
    var verdict = states.indexOf('no') >= 0 ? 'no' : (states.indexOf('unknown') >= 0 ? 'unknown' : 'candidate');
    return { verdict: verdict, items: items };
  }

  return {
    SIZES: SIZES, REGIONS: REGIONS, INDUSTRIES: INDUSTRIES, FACTS: FACTS,
    list: list, digits: digits, blank: blank, num: num, kstYear: kstYear, regionOf: regionOf,
    headcountAuto: headcountAuto, profileFromCompany: profileFromCompany, profileFromProspect: profileFromProspect,
    judge: judge
  };
});
```

```js
/* pu-govaid-view.js — 정부지원금 화면이 그릴 «묶음»을 만든다 (2026-10-11)
   설계: docs/superpowers/specs/2026-10-11-govaid-design.md (6절)
   ■ 순수 함수 — 자료를 받아 목록·묶음을 돌려줄 뿐, 읽기·쓰기·DOM 이 없다. 판정은 GovaidMatch 가 한다.
   ■ 업체와 이어진 예비 고객은 목록에 따로 세우지 않는다 — 업체의 aliases 로 접어,
     예비 고객 때 남긴 검토가 업체 화면에서도 보이게 한다(reviewOf). */
(function (root, factory) {
  if (typeof module === 'object' && module.exports) module.exports = factory(require('./pu-govaid-match.js'));
  else root.GovaidView = factory(root.GovaidMatch);
})(typeof self !== 'undefined' ? self : this, function (M) {
  'use strict';

  /* 푸른이알피 isActiveContract 의 «끝난» 상태와 같은 목록(js/rehab-match.js CLOSED_STATUS) */
  var CLOSED = ['closed', 'cancelled', 'transferred', 'terminated', 'expired'];
  function clean(v) { return v == null ? '' : String(v).trim(); }
  function byName(a, b) { return clean(a.name).localeCompare(clean(b.name), 'ko'); }
  function rows(v) {
    if (v && typeof v === 'object' && !Array.isArray(v) && Object.prototype.hasOwnProperty.call(v, 'v')) v = v.v;
    return M.list(v);
  }
  function programs(map) {
    return rows(map).filter(function (p) { return p && clean(p.id) && !p._deleted; }).sort(byName);
  }
  function companies(v) {
    return rows(v).filter(function (c) { return c && clean(c.id) && !c._deleted && CLOSED.indexOf(clean(c.status)) < 0; });
  }
  function subjects(companiesV, prospectsMap, pubCo, extrasMap) {
    pubCo = pubCo || {}; extrasMap = extrasMap || {};
    var byId = {}, out = [];
    companies(companiesV).forEach(function (co) {
      var p = M.profileFromCompany(co, pubCo[M.digits(co.bizNo)], extrasMap['co_' + clean(co.id)]);
      byId[p.id] = p; out.push(p);
    });
    rows(prospectsMap).forEach(function (pr) {
      if (!pr || !clean(pr.id) || pr._deleted) return;
      var linked = pr.companyLinkStatus === 'linked' && byId[clean(pr.companyId)];
      if (linked) { linked.aliases.push('pr_' + clean(pr.id)); return; }
      out.push(M.profileFromProspect(pr, pubCo[M.digits(pr.bizno)], extrasMap['pr_' + clean(pr.id)]));
    });
    return out.sort(byName);
  }
  function reviewOf(reviews, profile, pid) {
    reviews = reviews || {};
    var keys = [profile.key].concat(profile.aliases || []);
    for (var i = 0; i < keys.length; i++) {

    return g;

      else if (r.verdict === 'no') g.no++;

  }

    return { verdict: result.verdict, items: (result.items || []).map(function (i) { return { key: i.key, label: i.label, state: i.state }; }) };
```

```js
class Snap {
  constructor(v) { this.v = v; }
  child(p) { let c = this.v; for (const k of String(p).split('/')) c = c && typeof c === 'object' ? c[k] : undefined; return new Snap(c); }
  val() { return this.v === undefined ? null : this.v; }
  exists() { return this.v !== undefined && this.v !== null; }
}
const root = new Snap({ uid_roles: {
  staffUid: { status: 'active', sid: 'A-001' },
  retiredUid: { status: 'retired', sid: 'P-002' },
} });

function ev(expr, o) {
  o = o || {};
  const names = ['auth', 'root', 'data', 'newData', '$id', '$k', '$p'];
  const vals = [o.auth || null, root, new Snap(o.data), new Snap(o.newData), o.$id || '', o.$k || '', o.$p || ''];
  return Function(...names, '"use strict"; return Boolean(' + expr + ');')(...vals);
```
